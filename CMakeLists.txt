cmake_minimum_required(VERSION 3.20)
project(mm_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MME_BUILD_TESTS "Build unit tests" ON)
option(MME_BUILD_PYTHON "Build the pybind11 module" OFF)
set(MME_GTEST_DIR "" CACHE PATH "Local googletest source (skips download)")

add_compile_options(-Wall -Wextra -Wpedantic)
add_library(mme STATIC src/order_book.cpp src/options.cpp)
target_include_directories(mme PUBLIC include)
target_compile_options(mme PRIVATE $<$<CONFIG:Release>:-O3>)

add_executable(bench_book bench/bench_book.cpp)
target_link_libraries(bench_book PRIVATE mme)
add_executable(options_report bench/options_report.cpp)
target_link_libraries(options_report PRIVATE mme)

if(MME_BUILD_TESTS)
  enable_testing()
  if(MME_GTEST_DIR)
    add_subdirectory(${MME_GTEST_DIR} googletest EXCLUDE_FROM_ALL)
  else()
    include(FetchContent)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.15.2)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(mme_tests tests/test_order_book.cpp tests/test_options.cpp)
  target_link_libraries(mme_tests PRIVATE mme GTest::gtest_main)
  target_include_directories(mme_tests PRIVATE tests)
  include(GoogleTest)
  gtest_discover_tests(mme_tests)
endif()

if(MME_BUILD_PYTHON)
  find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND ${Python3_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_mme python/bindings.cpp)
  target_link_libraries(_mme PRIVATE mme)
endif()
